// 原地反转单链表（Iterative 迭代法）
public ListNode reverse(ListNode head) {
    ListNode prev = null;       // 前驱节点（初始为 null，表示反转后的尾）
    ListNode curr = head;       // 当前节点
    while (curr != null) {
        ListNode next = curr.next;  // 暂存下一个节点
        curr.next = prev;           // 当前节点指向前一个节点（反转方向）
        prev = curr;                // 前驱节点右移
        curr = next;                // 当前节点右移
    }
    return prev;   // prev 最终指向反转后的新头节点
}

public void bfs(TreeNode root) {
    if (root == null) return;

    Queue<TreeNode> q = new LinkedList<>();
    q.offer(root);

    while (!q.isEmpty()) {
        int size = q.size();  // 当前层节点数量

        for (int i = 0; i < size; i++) {
            TreeNode node = q.poll();

            // —— 在这里处理当前节点 node ——
            System.out.println(node.val);

            if (node.left != null) q.offer(node.left);
            if (node.right != null) q.offer(node.right);
        }
    }
}

class Solution {
    public int longestConsecutive(int[] nums) {
        int result=0;
        Set<Integer> set=new HashSet<>();
        //1.将所有数放到哈希集合
        for(int num:nums){
            set.add(num);
        }
        //2.遍历哈希集合，找前一个不存在的元素
        for(int num:set){
            if(!set.contains(num-1)){
                int count=1;
                //3.向右num+1扩展
                while(set.contains(num+1)){
                    count++;
                    num++;
                }
                result =Math.max(result,count);
            }
        }
        //4.返回结果
        return result;
    }
}

class Solution {
    public int longestConsecutive(int[] nums) {
        //先排序，后计算（虽然不是O(n)）
        
        if(nums.length==0) return 0;

        //1.设置结果
        int result=1;
        int count=1;
        //2.对数组进行排序
        Arrays.sort(nums);
        //3.遍历计算
        for(int i=0;i<nums.length-1;i++){
            if(nums[i]==nums[i+1]){
                count=count;
            }else if(nums[i]+1==nums[i+1]){
                count++;
                result =Math.max(count,result);
            }else{
                result =Math.max(count,result);
                count=1;//计数器重重为1
            }
        }
        return result;
    }
}

这个方法要注意数组越界问题，计数器设置问题，以及数组为空的情况。（提前进行排查）

-时间复杂度：O(nlogn)
-空间复杂度：O(n)

class Solution {
    public int maxArea(int[] height) {
        int left=0,right=height.length-1;
        int result=0;
        while(left<right){
            //看哪边大，哪边大，就移动另一边
            if(height[left]<height[right]){
                int v =Math.min(height[left],height[right])*(right-left);
                result=Math.max(v,result);
                left++;
            }else if(height[left]>=height[right]){
                int v =Math.min(height[left],height[right])*(right-left);
                result=Math.max(v,result);
                right--;
            }
        }
        //返回结果
        return result;
    }
}

class Solution {
    public List<List<Integer>> threeSum(int[] nums) {
        //1.优先判断边界情况
        if(nums.length<3) return new ArrayList<>();

        //2.创建返回模型;
        List<List<Integer>> list =new ArrayList<>();

        //3.对数组进行排序
        Arrays.sort(nums);

        //4.固定第一个数
        int n=nums.length;
        for(int i=0;i<n-2&&nums[i]<=0;i++){
            //5.对第一个数去重
            if(i>0 &&nums[i]==nums[i-1]) continue;

            int left =i+1;
            int right=n-1;

            //6.双指针
            while(left<right){
                int sum =nums[i]+nums[left]+nums[right];

                if(sum<0){
                    left++;
                }else if(sum>0){
                    right--;
                }else{
                    //7.找到第一个解
                    list.add(Arrays.asList(nums[i],nums[left],nums[right]));
                    //8.左指针去重
                    while(left<right && nums[left]==nums[left+1]) left++;
                    //9.右指针去重
                    while(left<right && nums[right]==nums[right-1]) right--;

                    //10.去重后同时移动
                    left++;
                    right--;
                }
            }

class Solution {
    public int trap(int[] height) {
        Stack<Integer> stack =new Stack<>(); 
        int res =0;

        for(int i=0;i<height.length;i++){

            //当前柱子比栈顶高<=>凹槽被右边界封住
            while(!stack.isEmpty() && height[i]>height[stack.peek()]){

                int bottom =stack.pop(); 
                if(stack.isEmpty()){
                    //没有左边界，不能接水
                    break;
                }

                int left =stack.peek(); //左边界
                int width =i-left -1;
                int h= Math.min(height[left],height[i])-height[bottom];

                res +=width *h;
            }

            //
            stack.push(i);
        }
        
        return res;
    }
}

class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        List<Integer> res =new ArrayList<>();
        char[] char_p=p.toCharArray();
        char[] char_s=s.toCharArray();

        if(char_s.length<char_p.length) return res;


        //1.统计p的字符频次
        int []freq =new int[26];
        for(char c:p.toCharArray()){
            freq[c-'a']++;
        }

        int left =0,right =0;
        int missing =p.length(); //还缺多少字符

        //2.滑动窗口
        while(right<char_s.length){
            //右边进入主窗口
            if(freq[char_s[right]-'a']>0){
                missing--;
            }

            freq[char_s[right]-'a']--;
            right++;


            //3.窗口大小等于p.length（）时开始进行判断
            if(right-left==p.length()){
                if(missing==0){
                    res.add(left);
                }

                //左边移出窗口
                if(freq[char_s[left]-'a']>=0){
                    missing++;
                }

                freq[char_s[left]-'a']++;
                left++;
            }
        }

        return res;

    }
}
